async function open(page: Page, level: 'basic' | 'business' | 'advanced') {
  await page.goto(`/libraries/ffmpeg-wasm?tab=demo&example=${level}`)
  const run = page.getByRole('button', { name: '运行当前示例', exact: true })
  if (await run.count()) await run.click()
  const app = page.getByRole('region', { name: 'FFmpeg wasm隔离转换实验', exact: true })
  await expect(app).toBeVisible()
  return app
}

function stereoWav() {
  const rate = 48000, channels = 2, samples = rate
  const bytes = Buffer.alloc(44 + samples * channels * 2)
  bytes.write('RIFF', 0); bytes.writeUInt32LE(bytes.length - 8, 4); bytes.write('WAVE', 8)
  bytes.write('fmt ', 12); bytes.writeUInt32LE(16, 16); bytes.writeUInt16LE(1, 20)
  bytes.writeUInt16LE(channels, 22); bytes.writeUInt32LE(rate, 24)
  bytes.writeUInt32LE(rate * channels * 2, 28); bytes.writeUInt16LE(channels * 2, 32)
  bytes.writeUInt16LE(16, 34); bytes.write('data', 36)
  bytes.writeUInt32LE(samples * channels * 2, 40)
  for (let sample = 0; sample < samples; sample++) {
    bytes.writeInt16LE(Math.round(Math.sin(2 * Math.PI * 440 * sample / rate) * 10000), 44 + sample * 4)
    bytes.writeInt16LE(Math.round(Math.sin(2 * Math.PI * 660 * sample / rate) * 10000), 46 + sample * 4)
  }
  return bytes
}

test('@E19 authorized original tone goes through real exec and downloaded PCM output', async ({ page }, info) => {
  const requests: string[] = []
  page.on('request', request => {
    if (/ffmpeg-core\.(js|wasm)(?:$|[?#])/.test(request.url())) requests.push(request.url())
  })
  const app = await open(page, 'basic')
  await expect(app.getByTestId('e19-command')).toContainText('-c:a pcm_s16le')
  await expect(app.getByTestId('e19-output-info')).toHaveCount(0)
  await app.getByRole('button', { name: '显式加载WASM引擎', exact: true }).click()
  await expect(app).toHaveAttribute('data-phase', 'ready', { timeout: 60_000 })
  await app.getByRole('button', { name: '执行转换', exact: true }).click()
  await expect(app).toHaveAttribute('data-phase', 'done', { timeout: 20_000 })
  await expect(app.getByTestId('e19-exit-code')).toHaveText('0')
  await expect(app.getByTestId('e19-cleanup')).toContainText('2/2')
  const output = JSON.parse(await app.getByTestId('e19-output-info').innerText())
  expect(output.sampleRate).toBe(16000); expect(output.channels).toBe(1)
  expect(output.seconds).toBeCloseTo(1, 3)
  const pendingDownload = page.waitForEvent('download')
  await app.getByRole('link', { name: '下载本次WAV', exact: true }).click()
  const download = await pendingDownload
  const path = info.outputPath('e19-original-tone.wav')
  await download.saveAs(path)
  const bytes = await readFile(path)
  expect(inspectPcmWav(bytes)).toEqual(output)
  expect(requests.some(url => url.endsWith('ffmpeg-core.wasm'))).toBe(true)
  expect(requests.every(url => new URL(url).origin === new URL(page.url()).origin)).toBe(true)
  // Save actual command, request URLs, Worker events and output WAV alongside browser evidence.
})

test('@E19 authorized loading cancellation closes the real Worker, then reloads and converts', async ({ page }) => {
  let closed = 0
  page.on('worker', worker => worker.on('close', () => { closed++ }))
  let release!: () => void, requested!: () => void
  const hold = new Promise<void>(resolve => { release = resolve })
  const reached = new Promise<void>(resolve => { requested = resolve })
  const pattern = '**/ffmpeg-core/ffmpeg-core.wasm'
  await page.route(pattern, async route => {
    requested()
    await hold
    // The first request may already be aborted by Worker termination. Never fulfill fake WASM.
    await route.continue().catch(() => {})
  })
  const app = await open(page, 'advanced')
  await app.getByRole('button', { name: '显式加载WASM引擎', exact: true }).click()
  await reached
  await app.getByRole('button', { name: '立即取消并终止Worker', exact: true }).click()
  await expect(app).toHaveAttribute('data-phase', 'cancelled')
  await expect.poll(() => closed).toBeGreaterThan(0)
  await expect(app.getByRole('button', { name: '执行转换', exact: true })).toBeDisabled()
  await expect(app.getByTestId('e19-audio')).toHaveCount(0)
  release()
  await page.unroute(pattern)
  await app.getByRole('button', { name: '显式加载WASM引擎', exact: true }).click()
  await expect(app).toHaveAttribute('data-phase', 'ready', { timeout: 60_000 })
  await app.getByRole('button', { name: '执行转换', exact: true }).click()
  await expect(app).toHaveAttribute('data-phase', 'done', { timeout: 20_000 })
  const parsed = JSON.parse(await app.getByTestId('e19-output-info').innerText())
  expect(parsed.seconds).toBe(0.5)
  await page.getByRole('button', { name: '停止并卸载', exact: true }).click()
  await expect(app).toHaveCount(0)
  // Also inspect the controller's actual Worker lifecycle observation for this unmount.
})
